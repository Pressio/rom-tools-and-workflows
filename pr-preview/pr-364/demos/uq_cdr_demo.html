Monte Carlo UQ for Convection-Diffusion-Reaction#

This example estimates the expected integrated right-boundary flux of the steady convection-diffusion-reaction (CDR) equation. Four uncertain inputs are sampled independently: advection magnitude, advection direction, diffusion coefficient, and reaction coefficient.

The standard Monte Carlo calculation uses the existing 21-by-21 CDR model. The multifidelity calculation treats that grid as high fidelity and a 9-by-9 grid as low fidelity. Four paired pilot samples estimate their correlation and select the final high- and low-fidelity counts under a budget of 12 high-fidelity-equivalent evaluations. The low-fidelity evaluation cost is set to 5% of a high-fidelity evaluation for this demonstration.

Run the example#

From the repository root:

python examples/uq_cdr_demo/example.py

Both workflows save their statistics, allocation metadata, samples, and model run directories below examples/uq_cdr_demo/uq_cdr_output.

Implementation#

 1"""Monte Carlo UQ for the steady convection-diffusion-reaction example."""
 2
 3import argparse
 4import os
 5from pathlib import Path
 6import sys
 7from typing import Optional
 8
 9import numpy as np
10
11from romtools.workflows.parameter_spaces import UniformParameterSpace
12from romtools.workflows.sampling_methods import MonteCarloSampler
13from romtools.workflows.uq import run_monte_carlo, run_multifidelity_monte_carlo
14
15
16EXAMPLE_DIRECTORY = Path(__file__).resolve().parent
17MODEL_DIRECTORY = EXAMPLE_DIRECTORY.parent / "models"
18if str(MODEL_DIRECTORY) not in sys.path:
19    sys.path.insert(0, str(MODEL_DIRECTORY))
20
21from steady_cdr_model import SteadyCdrQoiModel  # noqa: E402
22
23
24class IntegratedFluxCdrModel(SteadyCdrQoiModel):
25    """Return the integrated right-boundary flux as a scalar QoI."""
26
27    def compute_qoi(self, run_directory: str, parameter_sample: dict) -> np.ndarray:
28        with np.load(os.path.join(run_directory, "solution.npz")) as solution:
29            boundary_flux = solution["qoi"]
30        return np.array([np.sum(boundary_flux) * self.system.dy])
31
32
33def build_parameter_space() -> UniformParameterSpace:
34    """Construct independent uniform distributions for the CDR parameters."""
35    return UniformParameterSpace(
36        parameter_names=["bmag", "theta", "nu", "sigma"],
37        lower_bounds=[0.5, np.pi / 6.0, 0.02, 0.1],
38        upper_bounds=[1.5, np.pi / 3.0, 0.08, 0.6],
39        sampler=MonteCarloSampler,
40    )
41
42
43def main(output_directory: Optional[Path] = None) -> None:
44    """Run single- and multifidelity CDR mean-estimation examples."""
45    output_directory = (
46        EXAMPLE_DIRECTORY / "uq_cdr_output"
47        if output_directory is None
48        else Path(output_directory).resolve()
49    )
50    output_directory.mkdir(parents=True, exist_ok=True)
51    parameter_space = build_parameter_space()
52
53    mc_result = run_monte_carlo(
54        model=IntegratedFluxCdrModel(nx=21, ny=21),
55        parameter_space=parameter_space,
56        absolute_uq_directory=str(output_directory / "monte_carlo"),
57        number_of_samples=12,
58        random_seed=7,
59    )
60
61    mfmc_result = run_multifidelity_monte_carlo(
62        high_fidelity_model=IntegratedFluxCdrModel(nx=21, ny=21),
63        low_fidelity_model=IntegratedFluxCdrModel(nx=9, ny=9),
64        parameter_space=parameter_space,
65        absolute_uq_directory=str(output_directory / "multifidelity"),
66        pilot_sample_count=4,
67        high_fidelity_equivalent_budget=12.0,
68        low_to_high_fidelity_cost_ratio=0.05,
69        allocation_qoi_index=0,
70        random_seed=7,
71    )
72
73    print("CDR integrated right-boundary flux")
74    print(
75        f"MC:   mean={mc_result.mean[0]:.6e}, "
76        f"standard error={mc_result.standard_error[0]:.3e}"
77    )
78    print(
79        f"MFMC: mean={mfmc_result.mean[0]:.6e}, "
80        f"standard error={mfmc_result.standard_error[0]:.3e}, "
81        f"N_H={mfmc_result.high_fidelity_sample_count}, "
82        f"N_L={mfmc_result.low_fidelity_sample_count}"
83    )
84    print(f"Results written to {output_directory}")
85
86
87if __name__ == "__main__":
88    parser = argparse.ArgumentParser(description=__doc__)
89    parser.add_argument(
90        "--output-directory",
91        type=Path,
92        help="Output path (default: examples/uq_cdr_demo/uq_cdr_output)",
93    )
94    arguments = parser.parse_args()
95    main(arguments.output_directory)

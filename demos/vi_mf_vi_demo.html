VI and MF-VI Demo#

This example compares single-fidelity variational inference with multifidelity variational inference for a steady convection-diffusion-reaction model. Both methods approximate the posterior over the diffusion coefficient nu and reaction rate sigma with a Gaussian distribution.

The multifidelity method supplements its full-order-model samples with additional evaluations of a Gaussian-process surrogate. The surrogate is constructed automatically from the full-order-model parameter and QoI data, with both inputs and targets normalized before fitting.

Run the example#

The canonical implementation lives under examples/ and is exercised against the current romtools checkout in CI.

python examples/vi_mf_vi_demo/example.py

A reduced configuration is available for quick validation:

python examples/vi_mf_vi_demo/example.py --smoke

The full run uses 50 optimization iterations, eight FOM samples per iteration, and 64 additional GP-ROM samples for MF-VI. The smoke configuration reduces the grid, sample counts, and optimization iterations without changing the workflow API being tested.

Regenerate the published figures#

The Sphinx build does not execute scientific examples implicitly. Regenerate the checked-in figures explicitly after changes that affect this benchmark:

python examples/vi_mf_vi_demo/example.py \
    --output-dir docs/source/demos/notebooks

 30def _collect_vi_history(work_dir: Path):
 31    """Return the ELBO, variational-mean, and standard-deviation histories."""
 32    elbos = []
 33    iteration = 0
 34    while True:
 35        restart_path = work_dir / f"iteration_{iteration}" / "restart.npz"
 36        if not restart_path.exists():
 37            break
 38        with np.load(restart_path, allow_pickle=True) as data:
 39            elbos.append(float(data["elbo"]))
 40        iteration += 1
 41
 42    # history.npz stores moments in physical parameter coordinates. In
 43    # transformed bounded problems, exp(variational_log_std) from a restart is
 44    # an optimizer-coordinate quantity and must not be plotted as a physical
 45    # posterior standard deviation.
 46    with np.load(work_dir / "history.npz") as history:
 47        means = np.asarray(history["vi_history_variational_mean"], dtype=float)
 48        covariances = np.asarray(
 49            history["vi_history_variational_covariance"], dtype=float
 50        )
 51    standard_deviations = np.sqrt(
 52        np.maximum(np.diagonal(covariances, axis1=1, axis2=2), 0.0)
 53    )
 54    return np.asarray(elbos), means, standard_deviations

 57def _write_plots(
 58    output_dir: Path,
 59    parameter_names,
 60    truth: np.ndarray,
 61    vi_history,
 62    mf_vi_history,
 63) -> None:
 64    """Write ELBO and posterior-parameter convergence figures."""
 65    output_dir.mkdir(parents=True, exist_ok=True)
 66    vi_elbo, vi_means, vi_stds = vi_history
 67    mf_elbo, mf_means, mf_stds = mf_vi_history
 68
 69    figure, axis = plt.subplots(figsize=(6.5, 4.0))
 70    axis.plot(vi_elbo, marker="o", label="Single-fidelity VI")
 71    axis.plot(mf_elbo, marker="^", label="MF-VI with automatic GP ROM")
 72    axis.set(xlabel="Iteration", ylabel="ELBO", title="VI ELBO convergence")
 73    axis.grid(True, alpha=0.3)
 74    axis.legend()
 75    figure.tight_layout()
 76    figure.savefig(output_dir / "vi_mf_vi_elbo_convergence.png", dpi=180)
 77    plt.close(figure)
 78
 79    figure, axes = plt.subplots(
 80        1,
 81        len(parameter_names),
 82        figsize=(4.5 * len(parameter_names), 4.0),
 83        squeeze=False,
 84    )
 85    histories = (
 86        (vi_means, vi_stds, "Single-fidelity VI", "o"),
 87        (mf_means, mf_stds, "MF-VI with automatic GP ROM", "^"),
 88    )
 89    for parameter_index, axis in enumerate(axes.flat):
 90        for means, standard_deviations, label, marker in histories:
 91            iterations = np.arange(means.shape[0])
 92            axis.plot(iterations, means[:, parameter_index], marker=marker, label=label)
 93            axis.fill_between(
 94                iterations,
 95                means[:, parameter_index] - standard_deviations[:, parameter_index],
 96                means[:, parameter_index] + standard_deviations[:, parameter_index],
 97                alpha=0.2,
 98            )
 99        axis.axhline(truth[parameter_index], color="black", linestyle="--", label="Truth")
100        axis.set(
101            title=parameter_names[parameter_index],
102            xlabel="Iteration",
103            ylabel="Parameter value",
104        )
105        axis.grid(True, alpha=0.3)
106    axes.flat[0].legend()
107    figure.tight_layout()
108    figure.savefig(output_dir / "vi_mf_vi_parameter_convergence.png", dpi=180)
109    plt.close(figure)

112def main(
113    smoke: bool = False,
114    work_dir: Optional[str] = None,
115    output_dir: Optional[str] = None,
116) -> None:
117    """Run the VI/MF-VI comparison."""
118    grid_size = 10 if smoke else 25
119    sample_size = 4 if smoke else 8
120    rom_extra_sample_size = 4 if smoke else 64
121    max_iterations = 2 if smoke else 50
122
123    system = cdr.AdvectionDiffusionSystem(Nx=grid_size, Ny=grid_size)
124    model = CdrFomQoiModel(system, np.array([1.0, 1.0]))
125    truth_parameters = {"nu": 0.04, "sigma": 0.3}
126
127    root = Path(work_dir).resolve() if work_dir else Path(__file__).parent / "vi_mf_vi_work"
128    shutil.rmtree(root, ignore_errors=True)
129    truth_dir = root / "truth"
130    truth_dir.mkdir(parents=True)
131    model.populate_run_directory(str(truth_dir), truth_parameters)
132    model.run_model(str(truth_dir), truth_parameters)
133    observations = model.compute_qoi(str(truth_dir), truth_parameters)
134
135    parameter_mins = np.array([0.01, 0.1])
136    parameter_maxes = np.array([0.08, 0.6])
137    prior = GaussianParameterSpace(
138        ["nu", "sigma"],
139        (parameter_mins + parameter_maxes) / 2.0,
140        (parameter_maxes - parameter_mins) / 4.0,
141        MonteCarloSampler,
142    )
143    optimizer = VINewtonOptimizerConfig(
144        max_iterations=max_iterations,
145        gradient_norm_tolerance=0.0,
146        newton_hessian_type="full",
147        newton_curvature_strategy="lagged",
148        newton_hessian_averaging_factor=0.5,
149    )
150    line_search = VIStochasticNonmonotoneLineSearchConfig()
151    common_arguments = {
152        "model": model,
153        "prior_parameter_space": prior,
154        "observations": observations,
155        "observations_covariance": np.eye(observations.size) * 1.0e-5,
156        "parameter_mins": parameter_mins,
157        "parameter_maxes": parameter_maxes,
158        "optimizer_method": "newton",
159        "optimizer_config": optimizer,
160        "line_search_method": "stochastic_nonmonotone",
161        "line_search_config": line_search,
162        "restart_files_to_keep": max_iterations,
163        "baseline_method": "loo",
164        "random_seed": 1,
165    }
166
167    vi_dir = root / "vi"
168    mf_vi_dir = root / "mf_vi"
169    run_vi(
170        **common_arguments,
171        absolute_vi_directory=str(vi_dir),
172        sample_size=sample_size,
173        evaluation_concurrency=1,
174    )
175    mf_vi_with_auto_rom(
176        **common_arguments,
177        absolute_vi_directory=str(mf_vi_dir),
178        fom_sample_size=sample_size,
179        rom_extra_sample_size=rom_extra_sample_size,
180        fom_evaluation_concurrency=1,
181        rom_evaluation_concurrency=1,
182        rom_type="gp",
183        rom_args={
184            "normalize_parameters": True,
185            "normalize_targets": True,
186        },
187    )
188
189    output_path = Path(output_dir).resolve() if output_dir else Path(__file__).parent
190    truth = np.array([truth_parameters[name] for name in prior.get_names()])
191    _write_plots(
192        output_path,
193        prior.get_names(),
194        truth,
195        _collect_vi_history(vi_dir),
196        _collect_vi_history(mf_vi_dir),
197    )
198    print(f"Wrote VI/MF-VI figures to {output_path}")
